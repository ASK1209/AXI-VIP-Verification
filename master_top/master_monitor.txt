class master_monitor extends uvm_monitor;

        `uvm_component_utils (master_monitor)
        virtual axi_if.M_MON_MP vif;

        axi_xtn xtn, xtn1,xtn2,xtn3,xtn4;
        axi_xtn q1[$],q2[$];
        /*semaphore sem_awdc = new();
        semaphore sem_wdrc = new();
        semaphore sem_wdc = new(1);
        semaphore sem_awc = new(1);
        semaphore sem_wrc = new(1);
        semaphore sem_ardc = new();
        semaphore sem_arc = new(1);
        semaphore sem_rdc = new(1);*/
 semaphore sem_awc = new(1);    //semaphore for write address channel
        semaphore sem_wdc = new();      //semaphore for write data channel
        semaphore sem_wrc = new();      //semaphore for write response

        semaphore sem_arc = new(1);     //semaphore for read address channel
        semaphore sem_rdc = new();      //semaphore for read data channel

        master_config m_cfg;
        static int pkt_sent;
        uvm_analysis_port #(axi_xtn) monitor_port;

        extern function new(string name ="master_monitor",uvm_component parent);
        extern function void build_phase(uvm_phase phase);
        extern function void connect_phase(uvm_phase phase);
        extern task run_phase(uvm_phase phase);
        extern task collect_data();
        extern task collect_awaddr();
        extern task collect_wdata(axi_xtn xtn);
        extern task collect_bresp();
        extern task collect_raddr();
        extern task collect_rdata(axi_xtn xtn);
        extern function void report_phase(uvm_phase phase);
endclass

function master_monitor::new(string name ="master_monitor",uvm_component parent);
        super.new(name,parent);
	monitor_port=new("monitor_port",this);
endfunction

function void master_monitor::build_phase(uvm_phase phase);
        super.build_phase(phase);

        if(!uvm_config_db #(master_config)::get(this," ","master_config",m_cfg))
                `uvm_fatal("CONFIG","Cannot get() m_cfg from config_db. Have you set() it?")

endfunction

function void master_monitor::connect_phase(uvm_phase phase);
        super.connect_phase(phase);
        vif=m_cfg.vif;

endfunction

task master_monitor::run_phase(uvm_phase phase);
        forever
                collect_data();
//super.run_phase(phase);
endtask
task master_monitor::collect_data();
        fork
                begin
                        sem_awc.get(1);
                        collect_awaddr();
                        sem_wdc.put(1);
                        sem_awc.put(1);
                end
                begin
                        sem_wdc.get(1);
                        //sem_wdc.get(1);
                        collect_wdata(q1.pop_front());
                        sem_wrc.put(1);
                        sem_wdc.put(1);
                end
                begin
                        sem_wrc.get(1);
                        //sem_wrc.get(1);
                        collect_bresp();
			sem_wrc.put(1);
                end
                begin
                        sem_arc.get(1);
                        collect_raddr();
                        sem_rdc.put(1);
                        sem_arc.put(1);
                end
                begin
                        sem_rdc.get(1);
                        //sem_rdc.get(1);
                        collect_rdata(q2.pop_front());
                        sem_rdc.put(1);
                end
        join_any
endtask
task master_monitor::collect_awaddr();
xtn = axi_xtn::type_id::create("xtn");
wait(vif.m_mon_cb.AWVALID && vif.m_mon_cb.AWREADY)
        xtn.AWVALID = vif.m_mon_cb.AWVALID;
        xtn.AWADDR = vif.m_mon_cb.AWADDR;
        xtn.AWSIZE = vif.m_mon_cb.AWSIZE;
        xtn.AWID = vif.m_mon_cb.AWID;
        xtn.AWLEN = vif.m_mon_cb.AWLEN;
        xtn.AWBURST = vif.m_mon_cb.AWBURST;
        q1.push_back(xtn);
        monitor_port.write(xtn);
        pkt_sent++;
        `uvm_info("MST_MONITOR",$sformatf("printing from master monitor collect_awaddr \n %s",xtn.sprint()),UVM_LOW)
        @(vif.m_mon_cb);
endtask
task master_monitor::collect_wdata(axi_xtn xtn);
xtn1 = axi_xtn::type_id::create("xtn1");
xtn1 = xtn;
xtn.cal_addr();
xtn1.WDATA =  new[xtn.AWLEN +1];
xtn1.WSTRB = new[xtn.WDATA.size()];
foreach(xtn1.WDATA[i])
begin
        wait(vif.m_mon_cb.WVALID && vif.m_mon_cb.WREADY)
                xtn1.WSTRB[i] = vif.m_mon_cb.WSTRB;
		if(vif.m_mon_cb.WSTRB == 15)
                        xtn1.WDATA[i] = vif.m_mon_cb.WDATA;
                if(vif.m_mon_cb.WSTRB == 8)
                        xtn1.WDATA[i] = vif.m_mon_cb.WDATA[31:24];
                if(vif.m_mon_cb.WSTRB == 4)
                        xtn1.WDATA[i] = vif.m_mon_cb.WDATA[23:16];
                if(vif.m_mon_cb.WSTRB == 2)
                        xtn1.WDATA[i] = vif.m_mon_cb.WDATA[15:8];
                if(vif.m_mon_cb.WSTRB == 1)
                        xtn1.WDATA[i] = vif.m_mon_cb.WDATA[7:0];
                if(vif.m_mon_cb.WSTRB == 14)
                        xtn1.WDATA[i] = vif.m_mon_cb.WDATA[31:8];
                if(vif.m_mon_cb.WSTRB == 12)
                        xtn1.WDATA[i] = vif.m_mon_cb.WDATA[31:16];
                if(vif.m_mon_cb.WSTRB == 3)
                        xtn1.WDATA[i] = vif.m_mon_cb.WDATA[15:0];
                xtn1.WID = vif.m_mon_cb.WID;
                xtn1.WLAST = vif.m_mon_cb.WLAST;
                xtn1.WVALID = vif.m_mon_cb.WVALID;
                @(vif.m_mon_cb);
end
monitor_port.write(xtn1);
pkt_sent++;
`uvm_info("MST_MONITOR",$sformatf("printing from master monitor collect_wdata \n %s",xtn1.sprint()),UVM_LOW)
endtask
task master_monitor::collect_bresp();
xtn2 = axi_xtn::type_id::create("xtn2");
wait(vif.m_mon_cb.BREADY && vif.m_mon_cb.BVALID)
        xtn2.BRESP = vif.m_mon_cb.BRESP;
        monitor_port.write(xtn2);
        pkt_sent++;
        `uvm_info("MST_MONITOR",$sformatf("printing from master monitor collect_bresp \n %s",xtn2.sprint()),UVM_LOW)
        @(vif.m_mon_cb);
endtask
task master_monitor::collect_raddr();
xtn3 = axi_xtn::type_id::create("xtn3");
wait(vif.m_mon_cb.ARVALID && vif.m_mon_cb.ARREADY)
            xtn.ARVALID = vif.m_mon_cb.ARVALID;
       xtn.ARID = vif.m_mon_cb.ARID;
       xtn.ARLEN = vif.m_mon_cb.ARLEN;
       xtn.ARSIZE = vif.m_mon_cb.ARSIZE;
       xtn.ARBURST = vif.m_mon_cb.ARBURST;
        xtn.ARADDR = vif.m_mon_cb.ARADDR;
        q2.push_back(xtn3);
        @(vif.m_mon_cb);
        monitor_port.write(xtn3);
        pkt_sent++;
        `uvm_info("MST_MONITOR",$sformatf("printing from master monitor collect_raddr \n %s",xtn3.sprint()),UVM_LOW)
endtask
task master_monitor :: collect_rdata(axi_xtn xtn);
xtn4 = axi_xtn :: type_id :: create("xtn4");
xtn4 = xtn;
xtn4.cal_addr();
xtn4.RDATA = new[xtn.ARLEN+1];
foreach(xtn4.RDATA[i])
begin
        wait(vif.m_mon_cb.RVALID && vif.m_mon_cb.RREADY)

                xtn4.RDATA[i] = vif.m_mon_cb.RDATA;
    xtn4.RVALID = vif.m_mon_cb.RVALID;
                xtn4.RID = vif.m_mon_cb.RID;
    xtn4.RRESP = vif.m_mon_cb.RRESP;
                xtn4.RLAST = vif.m_mon_cb.RLAST;
                @(vif.m_mon_cb);
end
monitor_port.write(xtn4);
pkt_sent++;
`uvm_info("MST_MONITOR",$sformatf("printing from master monitor collect_rdata \n %s",xtn4.sprint()),UVM_LOW)
endtask
function void master_monitor :: report_phase(uvm_phase phase);
`uvm_info("MST_MONITOR",$sformatf("no : of packets sent are : %0d",pkt_sent),UVM_LOW)
endfunction
